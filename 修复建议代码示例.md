## 1. 修复 LocalStorage 数据丢失风险

### 方案：使用 IndexedDB 作为主存储

**创建文件**: `src/services/storageManager.js`

```javascript
/**
 * 统一存储管理器
 * 优先级: IndexedDB > localStorage > 内存
 */
class StorageManager {
  constructor() {
    this.db = null
    this.dbName = '91writing'
    this.storeName = 'novels'
    this.initDB()
  }

  // 初始化 IndexedDB
  async initDB() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(this.dbName, 1)
      
      request.onerror = () => reject(request.error)
      
      request.onsuccess = () => {
        this.db = request.result
        console.log('IndexedDB 初始化成功')
        resolve(this.db)
      }
      
      request.onupgradeneeded = (event) => {
        const db = event.target.result
        if (!db.objectStoreNames.contains(this.storeName)) {
          db.createObjectStore(this.storeName, { keyPath: 'id' })
        }
      }
    })
  }

  // 保存数据到 IndexedDB
  async save(key, data) {
    if (!this.db) await this.initDB()
    
    return new Promise((resolve, reject) => {
      const transaction = this.db.transaction([this.storeName], 'readwrite')
      const store = transaction.objectStore(this.storeName)
      
      const request = store.put({
        id: key,
        data: data,
        timestamp: Date.now()
      })
      
      request.onerror = () => reject(request.error)
      request.onsuccess = () => {
        // 同时保存到 localStorage 作为缓存
        try {
          localStorage.setItem(key, JSON.stringify(data))
        } catch (e) {
          console.warn('localStorage 满了:', e)
        }
        resolve(true)
      }
    })
  }

  // 从 IndexedDB 读取数据
  async get(key) {
    if (!this.db) await this.initDB()
    
    return new Promise((resolve, reject) => {
      const transaction = this.db.transaction([this.storeName], 'readonly')
      const store = transaction.objectStore(this.storeName)
      const request = store.get(key)
      
      request.onerror = () => reject(request.error)
      request.onsuccess = () => {
        if (request.result) {
          resolve(request.result.data)
        } else {
          // 降级到 localStorage
          try {
            const cached = localStorage.getItem(key)
            resolve(cached ? JSON.parse(cached) : null)
          } catch (e) {
            resolve(null)
          }
        }
      }
    })
  }

  // 自动备份到本地文件
  async autoBackup() {
    try {
      const allData = await this.getAllData()
      const backup = {
        timestamp: new Date().toISOString(),
        data: allData,
        version: '0.7.0'
      }
      
      // 自动保存到浏览器下载文件夹
      const blob = new Blob([JSON.stringify(backup, null, 2)], {
        type: 'application/json'
      })
      const url = URL.createObjectURL(blob)
      const link = document.createElement('a')
      link.href = url
      link.download = `91writing_backup_${Date.now()}.json`
      link.click()
      URL.revokeObjectURL(url)
    } catch (error) {
      console.error('备份失败:', error)
    }
  }

  // 数据恢复
  async restore(backupData) {
    try {
      if (!this.db) await this.initDB()
      
      const transaction = this.db.transaction([this.storeName], 'readwrite')
      const store = transaction.objectStore(this.storeName)
      
      // 清空现有数据
      store.clear()
      
      // 导入备份数据
      for (const item of backupData.data) {
        store.add(item)
      }
      
      return new Promise((resolve, reject) => {
        transaction.onerror = () => reject(transaction.error)
        transaction.oncomplete = () => resolve(true)
      })
    } catch (error) {
      console.error('数据恢复失败:', error)
      throw error
    }
  }

  // 获取所有数据
  async getAllData() {
    if (!this.db) await this.initDB()
    
    return new Promise((resolve, reject) => {
      const transaction = this.db.transaction([this.storeName], 'readonly')
      const store = transaction.objectStore(this.storeName)
      const request = store.getAll()
      
      request.onerror = () => reject(request.error)
      request.onsuccess = () => resolve(request.result)
    })
  }
}

export default new StorageManager()
```

```javascript
/**
 * 指数退避重试函数
 */
async retryWithBackoff(fn, maxRetries = 3, baseDelay = 1000) {
  let lastError
  
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn()
    } catch (error) {
      lastError = error
      
      // 最后一次重试失败，直接抛出
      if (i === maxRetries - 1) {
        throw error
      }
      
      // 计算延迟时间（指数退避）
      const delay = baseDelay * Math.pow(2, i) + Math.random() * 1000
      console.log(`第 ${i + 1} 次重试失败，${delay}ms 后重试...`)
      
      await new Promise(resolve => setTimeout(resolve, delay))
    }
  }
  
  throw lastError
}

/**
 * 改进的流式生成方法
 */
async generateTextStreamWithRetry(prompt, options = {}, onChunk = null) {
  const maxRetries = 3
  let retryCount = 0
  let savedContent = '' // 保存已生成的内容
  
  const attemptGeneration = async () => {
    try {
      return await this.generateTextStream(prompt, options, (chunk, fullContent) => {
        savedContent = fullContent // 保存当前内容
        if (onChunk) {
          onChunk(chunk, fullContent)
        }
      })
    } catch (error) {
      retryCount++
      
      // 网络错误且有已生成的内容，询问用户是否继续
      if (retryCount < maxRetries && savedContent.length > 0) {
        console.warn(`生成中断，已获得 ${savedContent.length} 字节内容，正在重试...`)
        ElMessage.warning(`生成中断，已保存 ${savedContent.length} 字节，正在重试...`)
        
        throw error // 继续重试
      }
      
      throw error
    }
  }
  
  try {
    return await this.retryWithBackoff(attemptGeneration, maxRetries, 1000)
  } catch (finalError) {
    // 如果最终失败，但有已生成的内容，返回它
    if (savedContent.length > 0) {
      ElMessage.warning(`生成中断，但已获得部分内容 (${savedContent.length} 字)`)
      return savedContent
    }
    
    throw finalError
  }
}
```

```javascript
async generateCharacter(theme, characterType = '') {
  try {
    const typeInfo = characterType ? `角色类型：${characterType}` : ''
    const prompt = `请根据主题"${theme}"生成一个小说人物...`
    
    const response = await this.generateTextStream(prompt, {}, null)
    
    // 尝试提取 JSON
    let parsed
    try {
      parsed = JSON.parse(response)
    } catch (parseError) {
      // 尝试从响应中提取 JSON 块
      const jsonMatch = response.match(/\{[\s\S]*\}/)
      if (jsonMatch) {
        parsed = JSON.parse(jsonMatch[0])
      } else {
        throw parseError
      }
    }
    
    // 验证必需字段
    const required = ['name', 'personality']
    const missing = required.filter(field => !parsed[field])
    
    if (missing.length > 0) {
      console.warn('AI响应缺少必要字段:', missing)
      // 返回部分数据 + 默认值
      return { ...this.getDefaultCharacter(), ...parsed }
    }
    
    return parsed
  } catch (error) {
    console.error('生成角色失败:', error)
    ElMessage.error('生成角色失败，请重试或手动填写')
    
    // 返回默认对象而非抛出错误
    return this.getDefaultCharacter()

}
```

```javascript
import { defineStore } from 'pinia'
import { reactive, watch, ref } from 'vue'

export const useNovelStore = defineStore('novel', () => {
  // 统一的小说数据结构
  const novelData = reactive({
    current: null,
    chapters: [],
    selectedChapter: null,
    characters: [],
    worldSettings: [],
    corpus: [],
    metadata: {
      apiConfig: {},
      goals: [],
      prompts: []
    }
  })

  // 当前编辑状态
  const editingState = reactive({
    isDirty: false,
    lastSaveTime: null,
    autoSaveInterval: null
  })

  // 初始化自动保存
  const initAutoSave = () => {
    // 监听数据变化
    watch(
      () => novelData,
      () => {
        editingState.isDirty = true
      },
      { deep: true }
    )

    // 启动自动保存定时器
    editingState.autoSaveInterval = setInterval(() => {
      if (editingState.isDirty && novelData.current) {
        saveCurrentNovel()
      }
    }, 5000)
  }

  // 更新章节状态（确保跨页面同步）
  const updateChapterStatus = (chapterId, status) => {
    const chapter = novelData.chapters.find(c => c.id === chapterId)
    if (chapter) {
      chapter.status = status
      chapter.updatedAt = new Date().toISOString()
      editingState.isDirty = true
      
      // 触发存储更新
      saveCurrentNovel()
    }
  }

  // 保存当前小说
  const saveCurrentNovel = async () => {
    if (!novelData.current) return
    
    try {
      // 更新聚合数据
      novelData.current.chapters = novelData.chapters
      novelData.current.characters = novelData.characters
      novelData.current.worldSettings = novelData.worldSettings
      novelData.current.updatedAt = new Date().toISOString()
      
      // 保存到存储
      await storageManager.save(`novel_${novelData.current.id}`, novelData.current)
      
      editingState.isDirty = false
      editingState.lastSaveTime = new Date()
      
      console.log('小说已自动保存')
    } catch (error) {
      console.error('保存小说失败:', error)
      ElMessage.error('自动保存失败')
    }
  }

  // 加载小说
  const loadNovel = async (novelId) => {
    try {
      const data = await storageManager.get(`novel_${novelId}`)
      if (data) {
        novelData.current = data
        novelData.chapters = data.chapters || []
        novelData.characters = data.characters || []
        novelData.worldSettings = data.worldSettings || []
        novelData.selectedChapter = novelData.chapters[0] || null
      }
    } catch (error) {
      console.error('加载小说失败:', error)
      throw error
    }
  }

  // 清理资源
  const cleanup = () => {
    if (editingState.autoSaveInterval) {
      clearInterval(editingState.autoSaveInterval)
    }
  }

  return {
    novelData,
    editingState,
    initAutoSave,
    updateChapterStatus,
    saveCurrentNovel,
    loadNovel,
    cleanup
  }
})
```

**创建加密工具**: `src/utils/encryption.js`

```javascript
/**
 * 使用 Web Crypto API 进行加密
 */
class EncryptionUtil {
  // 生成加密密钥
  async generateKey() {
    return await crypto.subtle.generateKey(
      { name: 'AES-GCM', length: 256 },
      true, // 可导出
      ['encrypt', 'decrypt']
    )
  }

  // 加密数据
  async encrypt(data, key) {
    const encoder = new TextEncoder()
    const iv = crypto.getRandomValues(new Uint8Array(12))
    
    const encrypted = await crypto.subtle.encrypt(
      { name: 'AES-GCM', iv },
      key,
      encoder.encode(JSON.stringify(data))
    )

    return {
      iv: Array.from(iv),
      data: Array.from(new Uint8Array(encrypted))
    }
  }

  // 解密数据
  async decrypt(encrypted, key) {
    const decoder = new TextDecoder()
    const iv = new Uint8Array(encrypted.iv)
    const data = new Uint8Array(encrypted.data)

    const decrypted = await crypto.subtle.decrypt(
      { name: 'AES-GCM', iv },
      key,
      data
    )

    return JSON.parse(decoder.decode(decrypted))
  }

  // 存储加密后的 API 密钥
  async storeEncryptedAPIKey(apiKey) {
    try {
      const key = await this.generateKey()
      const encrypted = await this.encrypt({ key: apiKey }, key)
      
      // 保存加密密钥和加密数据
      localStorage.setItem('api_key_encrypted', JSON.stringify(encrypted))
      
      // 保存主密钥到内存（应用运行期间有效）
      sessionStorage.setItem('master_key', await this.exportKey(key))
      
      return true
    } catch (error) {
      console.error('加密存储失败:', error)
      return false
    }
  }

  // 导出密钥
  async exportKey(key) {
    const exported = await crypto.subtle.exportKey('jwk', key)
    return JSON.stringify(exported)
  }

  // 导入密钥
  async importKey(keyData) {
    return await crypto.subtle.importKey(
      'jwk',
      JSON.parse(keyData),
      { name: 'AES-GCM', length: 256 },
      true,
      ['encrypt', 'decrypt']
    )
  }
}

export default new EncryptionUtil()
```

```vue
const router = useRouter()
const showGuide = ref(false)
const currentStep = ref(0)

onMounted(() => {
  // 检查是否是首次访问
  const hasSeenGuide = localStorage.getItem('onboarding_completed')
  if (!hasSeenGuide) {
    showGuide.value = true
  }
})

const nextStep = () => {
  if (currentStep.value < 3) {
    currentStep.value++
  }
}

const previousStep = () => {
  if (currentStep.value > 0) {
    currentStep.value--
  }
}

const goToApiConfig = () => {
  router.push('/settings')
}

const goToNovels = () => {
  router.push('/novels')
}

const completeGuide = () => {
  localStorage.setItem('onboarding_completed', 'true')
  showGuide.value = false
}

const closeGuide = () => {
  showGuide.value = false
}
```
